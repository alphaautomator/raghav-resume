% Custom commands
% Note: no space before \vspace, and % guards on every line end.
% A trailing space here makes a full-width bullet wrap onto an empty line.
\newcommand{\resumeItem}[1]{%
  \item\small{#1}%
  \vspace{-2pt}%
}

\newcommand{\resumeItemnew}[1]{%
  \item\small{#1}%
  \vspace{-6pt}%
}

\newcommand{\resumeItemHead}[1]{
  \textbf{#1} \vspace{-8pt}
}

\newcommand{\resumeCustomItem}[1]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1}
    \end{tabular*}\vspace{-3pt}
}


\newcommand{\resumeCustomItemOne}{\vspace{4pt}}

      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeSubheadingnew}[4]{
